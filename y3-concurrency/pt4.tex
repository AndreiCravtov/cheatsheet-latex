\begin{lstlisting}[language=C++, escapechar=\$]
// `SYS_futex` is a SYSCALL with many OPERATIONS
int futex(uint32_t *p, int op, uint32_t v, const struct timespec *timeout, uint32_t *p2, uint32_t v3) {
  return syscall(SYS_futex, p, op, v, timeout, p2, v3);
}
// `FUTEX_WAIT` operation:
//   - IF `*p != v` THEN sets `errno=EAGAIN` and returns `-1` immediately
//   - OTHERWISE puts thread to sleep & adds it to KERNEL SPACE wait queue associated with `p`; returns `0` when woken up
int futex_wait(uint32_t *p, uint32_t v) {
  return futex(p, FUTEX_WAIT, v, nullptr, nullptr, 0);
}
// `FUTEX_WAKE` operation:
//   - Wake up at most `wake_count` threads in `p`-associated wait queue; returns actual number of woken up threads
//   - `wake_count` most commonly `1` or `INT_MAX`
int futex_wake(uint32_t *p, uint32_t wake_count) {
  return futex(p, FUTEX_WAKE, wake_count, NULL, NULL, 0);
}
\end{lstlisting}

\stb{Simple implementation}
\begin{lstlisting}[language=C++, escapechar=\$]
struct mutex_simple {
  const int kFree = 0; const int kLocked = 1;
  std::atomic<int> state_ = {kFree};
  void lock() {
    while (state_.exchange(kLocked) == kLocked) {
      futex_wait(&state, kLocked);
    }
  }
  void unlock() {
    state_.store(kFree);
    futex_wake(&state, 1);  // wake up 1 waiter
  }
};
\end{lstlisting}

\begin{lstlisting}[language=C++, escapechar=\$]
class mutex_smarter {
  const int kFree = 0;
  const int kLockedNoWaiters = 1;
  const int kLockedWaiters = 2;
  void lock() {
    int old = cmpxchg(kFree, kLockedNoWaiters);
    if (old == kFree) return;  // got lock w/out contention
    
    do {
      if (old == kLockedWaiters || cmpxchg(kLockedNoWaiters, kLockedWaiters) != kFree) {
        futex_wait(&state_, kLockedWaiters);
      }
      old = cmpxchg(kFree, kLockedWaiters);
    } while (old != kFree);
  }
  void unlock() {
    if (state_.exchange(kFree) == kLockedWaiters) {
      futex_wake(&state_, 1);
    }
  }
 private:
  std::atomic<int> state_ = {kFree};
  int cmpxchg(int expected, int desired) {
    int e = expected;  // mutable stack variable
    state_.compare_exchange_strong(e, desired);
    return e;
  }
};
\end{lstlisting}
