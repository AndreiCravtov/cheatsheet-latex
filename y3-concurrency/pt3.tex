\begin{lstlisting}[language=C++, escapechar=\$]
struct ticket_lock {
  std::atomic<size_t> next_ = {0};
  std::atomic<size_t> curr_ = {0};
  void lock() {
    auto ticket = next_.fetch_add(1, std::relaxed);
    while (curr_.load(std::acquire) != ticket) {
      _mm_pause();
    }
  }
  void unlock() {
    auto current = curr_.load(std::relaxed);
    curr_.store(current + 1, std::release);
  }
};
\end{lstlisting}
\stub{NOTE}: performs worse than spinlock, but more fair.
