class FifoQueue {
  // ...
  void Enq(const int value) {
    std::unique_lock<std::mutex> lock(mutex_);
    $\inlinebox{not\_full\_.wait}$(lock, [this]() -> bool { return count_ < elements_.size(); });
    elements_[tail_] = value;
    tail_ = (tail_ + 1) % elements_.size();
    count_++;
    $\inlinebox{not\_empty\_.notify\_one();}$
  }
  int Deq() {
    std::unique_lock<std::mutex> lock(mutex_);
    $\inlinebox{not\_empty\_.wait}$(lock, [this]() -> bool { return count_ != 0; });
    int result = elements_[head_];
    head_ = (head_ + 1) % elements_.size();
    count_--;
    $\inlinebox{not\_full\_.notify\_one();}$
    return result;
  }
  // ...
};
