\subsubsection*{4b - Concurrency in Rust}

\begin{lstlisting}[language=C++]
// can only capture 'static refs
let h = thread::spawn(|| -> T {...});
h.join().unwrap(); // remember to join handle

// attach condvar to data with this
let p: Arc<(Mutex<T>, Condvar)> = ...;
let (m, cv) = &*p; // unpack mutex & condvar
cv.notify_one/all(); // condvar notify
let mut guard = m.lock().unwrap();
let val = &mut *guard; // ASIDE: mut deref
while /*actual cond.*/ { // wait for condvar
  guard = cv.wait(guard).unwrap();
}

\end{lstlisting}
